M200:=MA(CLOSE,200);
M250:=MA(CLOSE,250);
TOUCH:=REF(CLOSE,5)>REF(M200,5) AND REF(LOW,4)<=REF(M200,4);
MAUP:=M200>REF(M200,1) AND REF(M200,1)>REF(M200,2) AND REF(M200,2)>REF(M200,3) AND REF(M200,3)>REF(M200,4) AND REF(M200,4)>REF(M200,5);
HIGHOVER:=REF(HIGH,3)>REF(M200,3) AND REF(HIGH,2)>REF(M200,2) AND REF(HIGH,1)>REF(M200,1) AND HIGH>M200;
LONGFILTER:=REF(M200,4)>REF(M250,4) AND REF(M200,3)>REF(M250,3) AND REF(M200,2)>REF(M250,2) AND REF(M200,1)>REF(M250,1) AND M200>M250;
XG:TOUCH AND MAUP AND LONGFILTER AND HIGHOVER AND CLOSE>M200;
