M120:=MA(CLOSE,120);
TOUCH:=REF(CLOSE,5)>REF(M120,5) AND REF(LOW,4)<=REF(M120,4);
MAUP:=M120>REF(M120,1) AND REF(M120,1)>REF(M120,2) AND REF(M120,2)>REF(M120,3) AND REF(M120,3)>REF(M120,4) AND REF(M120,4)>REF(M120,5);
HIGHOVER:=REF(HIGH,3)>REF(M120,3) AND REF(HIGH,2)>REF(M120,2) AND REF(HIGH,1)>REF(M120,1) AND HIGH>M120;
XG:TOUCH AND MAUP AND HIGHOVER AND CLOSE>M120;
